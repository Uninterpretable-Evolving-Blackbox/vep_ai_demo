### 3. Generated VEP Command

--plugin ClinVar \
--plugin dbNSFP \
--plugin CADD \
--plugin SpliceAI \
--plugin AlphaMissense \
--plugin EVE \
--plugin Phenotypes \
--plugin MaveDB \
--plugin Mastermind \
--mane --appris --tsl --symbol --hgvs --protein --biotype --numbers --domains
